func TestStellarSwirlReusesItsVortexAndDetonatesAccumulatedStacksOnce(t *testing.T) {
 reactions,detonations,maxVortices:=0,0,0
 stacks:=[]int{}
 source:="options duration=6 workers=1 iteration=1;\n"+
  "target lvl=90 resist=0.1;\n"+
  "sandrone char lvl=90/90 cons=0 talent=10,10,10;\n"+
  "sandrone add weapon=\"wastergreatsword\" refine=1 lvl=70/70;\n"+
  "sucrose char lvl=90/90 cons=1 talent=6,6,6;\n"+
  "sucrose add weapon=\"apprenticesnotes\" refine=1 lvl=70/70;\n"+
  "active sandrone; sandrone skill; sucrose skill; sucrose skill; wait(360);"
 runScenario(t,source,func(c *core.Core){
  c.Events.Subscribe(event.OnStellarSwirl,func(args ...any){
   reactions++
   c.Tasks.Add(func(){
    count:=0
    for _,g:=range c.Combat.Gadgets(){
     if _,ok:=g.(*reactable.StellarVortex);ok {
      count++
      if g.GadgetTyp()!=info.GadgetTypStellarVortex{t.Error("vortex registered as another gadget type")}
     }
    }
    maxVortices=max(maxVortices,count)
    stacks=append(stacks,int(c.Flags.Custom[reactable.StellarSwirlKey+"-stacks"]))
   },1)
  },"test-vortex-reuse")
  c.Events.Subscribe(event.OnStellarVortexDetonate,func(args ...any){detonations++},"test-vortex-detonation")
 })
 if reactions!=2||maxVortices!=1||detonations!=1{t.Fatalf("reactions=%d maxVortices=%d detonations=%d",reactions,maxVortices,detonations)}
 if len(stacks)!=2||stacks[0]!=1||stacks[1]!=2{t.Fatalf("stack progression=%v",stacks)}
}

func TestStellarSwirlFullStacksDetonateEarlyAndAllowANewCycle(t *testing.T) {
 reactions:=0
 detonationFrames:=[]int{}
 reactionFrames:=[]int{}
 source:="options duration=8 workers=1 iteration=1;\n"
 for i:=0;i<6;i++{source+="target lvl=90 resist=0.1;\n"}
 source+=`sandrone char lvl=90/90 cons=0 talent=10,10,10;
sandrone add weapon="wastergreatsword" refine=1 lvl=70/70;
sucrose char lvl=90/90 cons=1 talent=6,6,6;
sucrose add weapon="apprenticesnotes" refine=1 lvl=70/70;
active sandrone;
sandrone skill; sucrose skill; sucrose skill; wait(480);`
 runScenario(t,source,func(c *core.Core){
  c.Events.Subscribe(event.OnStellarSwirl,func(args ...any){
   reactions++
   reactionFrames=append(reactionFrames,c.F)
  },"test-full-vortex-reactions")
  c.Events.Subscribe(event.OnTick,func(args ...any){
   raw:=c.Flags.Custom[reactable.StellarSwirlKey+"-stacks"]
   if raw<0||raw>6{t.Errorf("raw stack storage exceeded cap: %v",raw)}
  },"test-raw-stack-cap")
  c.Events.Subscribe(event.OnStellarVortexDetonate,func(args ...any){
   detonationFrames=append(detonationFrames,c.F)
   if c.Flags.Custom[reactable.StellarSwirlKey+"-stacks"]!=0{t.Error("detonation did not reset stacks")}
  },"test-full-vortex-detonation")
 })
 if reactions!=12||len(detonationFrames)!=2{t.Fatalf("reactions=%d detonation frames=%v",reactions,detonationFrames)}
 for i,f:=range detonationFrames {
  if f!=reactionFrames[i*6+5]{t.Errorf("cycle %d exploded at %d instead of sixth reaction at %d",i,f,reactionFrames[i*6+5])}
 }
}

func TestRealFieldKeepsSixFrameStackICD(t *testing.T) {
 scenario(t,0,"wait(12);",func(c *core.Core){
  for _,hit:=range []struct{frame int;ele attributes.Element}{{1,attributes.Electro},{2,attributes.Cryo},{3,attributes.Cryo},{4,attributes.Cryo},{9,attributes.Cryo},{10,attributes.Cryo}} {
   ai:=info.AttackInfo{ActorIndex:0,Abil:"field stack test",AttackTag:attacks.AttackTagExtra,Element:hit.ele,Durability:25,ICDTag:attacks.ICDTagNone,ICDGroup:attacks.ICDGroupDefault,Mult:1}
   c.QueueAttack(ai,combat.NewCircleHitOnTarget(c.Combat.PrimaryTarget(),nil,3),hit.frame,hit.frame)
  }
  c.Tasks.Add(func(){if n:=c.Flags.Custom[reactable.PolestarFieldKey+"-recorded"];n!=1{t.Errorf("same ICD window recorded %v stacks",n)}},5)
  // The pinned modifier expiry is inclusive: an entry at3 with duration6
  // remains active at9 and becomes available at10.
  c.Tasks.Add(func(){if n:=c.Flags.Custom[reactable.PolestarFieldKey+"-recorded"];n!=1{t.Errorf("expiry boundary recorded %v stacks",n)}},9)
  c.Tasks.Add(func(){if n:=c.Flags.Custom[reactable.PolestarFieldKey+"-recorded"];n!=2{t.Errorf("next ICD window recorded %v stacks",n)}},11)
 })
}

func TestC6CycleHasFourExtraBeamsAndSharedC4Cooldown(t *testing.T) {
 originals,extras:=0,0;coordinated:=[]int{}
 scenario(t,6,"sandrone charge;",func(c *core.Core){
  addElectro(c)
  c.Events.Subscribe(event.OnEnemyHit,func(args ...any){
   ai:=args[1].(*info.AttackEvent).Info
   if ai.AttackTag!=attacks.AttackTagDirectStellarConduct{return}
   if math.Abs(ai.Elevation-0.2)>1e-9{t.Errorf("C6 elevation=%v",ai.Elevation)}
   switch ai.Abil {
   case "Condensed Beam":
    originals++
    if originals==1&&math.Abs(c.Player.ByIndex(0).ReactBonus(ai)-0.3)>1e-9{t.Error("C1 reaction bonus missing during decoding")}
    if math.Abs(ai.Mult-1.957)>1e-9{t.Errorf("C3 level13 multiplier=%v",ai.Mult)}
   case "Condensed Cluster Beam":
    extras++;if ai.Mult!=0.8||ai.Durability!=0{t.Errorf("C6 extra beam=%+v",ai)}
   case "Prismatic Resonance Cannon (C4)":
    coordinated=append(coordinated,c.F);if ai.Mult!=1.25||ai.Durability!=0{t.Errorf("C4 attack=%+v",ai)}
   }
  },"test-c6")
  c.Tasks.Add(func(){if c.Player.ByIndex(0).ReactBonus(info.AttackInfo{AttackTag:attacks.AttackTagDirectStellarConduct})!=0{t.Error("C1 bonus persisted after decoding")}},410)
 })
 if originals!=6||extras!=4{t.Fatalf("original=%d extra=%d",originals,extras)}
 if len(coordinated)!=2||coordinated[1]-coordinated[0]<240{t.Fatalf("C4 hit frames=%v",coordinated)}
}

func scenario(t *testing.T, cons int, actions string, attach func(*core.Core)) {
 t.Helper()
 scenarioTargets(t,cons,1,actions,attach)
}

func scenarioTargets(t *testing.T, cons, targets int, actions string, attach func(*core.Core)) {
 t.Helper()
 source:=fmt.Sprintf(`options duration=12 workers=1 iteration=1;
sandrone char lvl=90/90 cons=%d talent=10,10,10;
sandrone add weapon="wastergreatsword" refine=1 lvl=70/70;
active sandrone;
%s
wait(700);`,cons,actions)
 for i:=0;i<targets;i++{source="target lvl=90 resist=0.1;\n"+source}
 runScenario(t,source,attach)
}

func TestMultipleTargetsShareParticlesAndC4Cooldown(t *testing.T) {
 for _,targets:=range []int{1,3} {
  t.Run(fmt.Sprintf("targets%d",targets),func(t *testing.T){
   particles:=0
   cannonFrames:=map[int]int{}
   scenarioTargets(t,6,targets,"sandrone charge;",func(c *core.Core){
    addElectro(c)
    c.Events.Subscribe(event.OnParticleReceived,func(args ...any){particles++},"test-shared-particles")
    c.Events.Subscribe(event.OnEnemyHit,func(args ...any){
     if args[1].(*info.AttackEvent).Info.Abil=="Prismatic Resonance Cannon (C4)"{cannonFrames[c.F]++}
    },"test-shared-c4")
   })
   if particles!=2{t.Errorf("particle batches=%d want2 independent of target count",particles)}
   if len(cannonFrames)!=2{t.Fatalf("C4 frames=%v",cannonFrames)}
   first,last:=10000,0
   for frame,hits:=range cannonFrames{
    first=min(first,frame);last=max(last,frame)
    if hits!=targets{t.Errorf("C4 at %d hit %d targets, want%d",frame,hits,targets)}
   }
   if last-first<240{t.Errorf("shared C4 cooldown violated: %v",cannonFrames)}
  })
 }
}

func runScenario(t *testing.T, source string, attach func(*core.Core)) {
 t.Helper()
 file:=ast.NewFile()
 cfg,program,err:=parser.New(file,source).Parse();if err!=nil {t.Fatal(err)}
 if len(cfg.Errors)>0 {t.Fatal(cfg.Errors)}
 c,err:=simulation.NewCore(17,false,cfg);if err!=nil {t.Fatal(err)}
 evaluator,err:=eval.NewEvaluator(file,program,c);if err!=nil {t.Fatal(err)}
 sim,err:=simulation.New(cfg,evaluator,c);if err!=nil {t.Fatal(err)}
 if attach!=nil {attach(c)}
 result,err:=sim.Run();if err!=nil {t.Fatal(err)}
 if result.DPS<=0 {t.Fatalf("no actual damage: %v",result.DPS)}
}
